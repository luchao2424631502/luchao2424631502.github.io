[POJ 2524](http://poj.org/problem?id=2524)  
并查集模板题

```arduino
#include <iostream>
using namespace std;
const int maxn = 5e4+10;
int pre[maxn];
int n,m;
void init(int k) {
	for (int i=1; i<=k; i++) pre[i] = i; 
}
int find(int x) {
	if (x == pre[x]) return x;
	return pre[x] = find(pre[x]);
}
void merge(int a,int b) {
	int ra = find(a);
	int rb = find(b);
	if (ra == rb) return;
	pre[ra] = rb;
}
int a,b;
int cnt;
int main() {
//	freopen("a.txt","r",stdin);
	while (scanf("%d%d",&n,&m)) {
		if (n==0 && m==0) break;
		init(n); //并查集不能忘记初始化
		for (int i=1; i<=m; i++) {
			scanf("%d%d",&a,&b);
			merge(a,b);
		}
		int ans = 0;
		for (int i=1; i<=n; i++) 
			if (pre[i] == i) ans++;
		printf("Case %d: %d\n",++cnt,ans);
	}
	return 0;
}
```
