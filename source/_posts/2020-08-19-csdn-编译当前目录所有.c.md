头文件都放在`./include`中

```
CC                 = gcc
CFLAGS            = -o2 -I include/ 
SRC                = $(wildcard *.c)                        #代表所有的.c文件
TAR                = $(patsubst %.c,%, $(SRC)) #模式字符串替换

all: clean $(TAR)

$(TAR): %:%.c 
    $(CC) $< $(CFLAGS) -o $@

.PHONY:
    all clean 

clean:
    rm -f $(TAR)
```
