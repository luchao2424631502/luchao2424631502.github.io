~~这么简单的我都想不出来~~  
思路：

- 开一个`vis[]`标记在区间`[l,r]`中已经出现的画的数量,当每个画`vis[i]==1`那么看到的画的数量`++`,左端如果画的数量`>1`那么向左移动,
- 右端不断标记然后加入队列中，当 `画的数量=画家的数量` 记录区间端点值

```cpp
#include <iostream>
using namespace std;
const int maxn = 1e6+10;
int vis[2010];
int num[maxn];
int main () {
	freopen("a.txt","r",stdin);
	int n,m;
	cin >> n >> m;
	for (int i=1; i<=n; i++) 
		scanf("%d",&num[i]);

	int ans=0x7fffffff;
	int ans_l=1, ans_r=n;
	int left=1,len=0;

	for (int i = 1; i <= n; ++i) {
		vis[num[i]]++;
		if (vis[num[i]] == 1) len++;
		while (vis[num[left]]>1) {//左端的数出现超过2次，那么可以往右移动
			vis[num[left]]--;
			left++;
		} 
		if ( len==m && ans>i-left+1 ) {
			ans = i-left+1;
			ans_l = left;
			ans_r = i;
		}
	}
	cout << ans_l << " " << ans_r;
	return 0;
}
```
